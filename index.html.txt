        本週沒做滿 3 次，下一週晚餐只能吃<br>

        <button class="reset-btn" onclick="resetData()">重置所有紀錄</button>

<script>
    const totalWeeks = 8; // 預設提供 8 週紀錄表
    const players = ['芷昕', '嘉晏'];

    // 從 LocalStorage 載入資料，若無則初始化
    let store = JSON.parse(localStorage.getItem('workout_tracker_data')) || {};

    function initData() {
        for (let w = 1; w <= totalWeeks; w++) {
            if (!store[w]) {
                store[w] = { '芷昕': 0, '嘉晏': 0 };
            }
        }
    }

    function saveData() {
        localStorage.setItem('workout_tracker_data', JSON.stringify(store));
    }

    function toggleCheck(week, player, count) {
        // 如果點擊已勾選的最後一個，就取消；否則設定為點擊的次數
        if (store[week][player] === count) {
            store[week][player] = count - 1;
        } else {
            store[week][player] = count;
        }
        saveData();
        render();
    }

    function render() {
        const container = document.getElementById('weeks-container');
        container.innerHTML = '';

        for (let w = 1; w <= totalWeeks; w++) {
            const weekData = store[w];
            const card = document.createElement('div');
            card.className = 'week-card';

            let playersHtml = '';
            players.forEach(p => {
                const count = weekData[p] || 0;
                const isPass = count >= 3;
                
                playersHtml += `
                    <div class="player-box">
                        <div class="player-name">${p}</div>
                        <div class="check-group">
                            <div class="check-btn ${count >= 1 ? 'active' : ''}" onclick="toggleCheck(${w}, '${p}', 1)">✓</div>
                            <div class="check-btn ${count >= 2 ? 'active' : ''}" onclick="toggleCheck(${w}, '${p}', 2)">✓</div>
                            <div class="check-btn ${count >= 3 ? 'active' : ''}" onclick="toggleCheck(${w}, '${p}', 3)">✓</div>
                        </div>
                        <div class="status-tag ${isPass ? 'status-pass' : 'status-fail'}">
                            ${isPass ? '🎉 已達標' : `還差 ${3 - count} 次`}
                        </div>
                    </div>
                `;
            });

            card.innerHTML = `
                <div class="week-header">
                    <span class="week-title">WEEK ${w}</span>
                </div>
                <div class="players-grid">
                    ${playersHtml}
                </div>
            `;

            container.appendChild(card);
        }
    }

    function resetData() {
        if (confirm('確定要清空所有運動紀錄嗎？')) {
            localStorage.removeItem('workout_tracker_data');
            store = {};
            initData();
            render();
        }
    }

    initData();
    render();
</script>
